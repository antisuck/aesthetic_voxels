local function getRoot(character)
    if not character then
        return nil
    end

    return character:FindFirstChild("HumanoidRootPart") or character.PrimaryPart
end

local function getHumanoid(character)
    if not character then
        return nil
    end

    return character:FindFirstChildOfClass("Humanoid")
end

local function getCharacterPosition(character)
    if not character then
        return nil
    end

    local root = getRoot(character)

    if root then
        return root.Position
    end

    local ok, pivot = pcall(function()
        return character:GetPivot()
    end)

    if ok and pivot then
        return pivot.Position
    end

    return nil
end

local function getAimPart(character)
    if not character then
        return nil
    end

    local head = character:FindFirstChild("Head")

    if head and head:IsA("BasePart") then
        return head
    end

    local root = character:FindFirstChild("HumanoidRootPart")

    if root and root:IsA("BasePart") then
        return root
    end

    if character.PrimaryPart and character.PrimaryPart:IsA("BasePart") then
        return character.PrimaryPart
    end

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") and object.Transparency < 1 then
            return object
        end
    end

    return nil
end

local function isTeammate(player)
    if player == LocalPlayer then
        return true
    end

    local myTeam = LocalPlayer.Team

    return myTeam ~= nil and player.Team == myTeam
end

local function getPlayerFromCharacter(character)
    if not character then
        return nil
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character == character then
            return player
        end
    end

    return nil
end

local function isTargetable(player)
    if player == LocalPlayer then
        return false
    end

    if State.TeamCheck and isTeammate(player) then
        return false
    end

    local humanoid = getHumanoid(player.Character)

    return humanoid ~= nil and humanoid.Health > 0
end

local function isLocalPlayerJumping()
    local humanoid = getHumanoid(LocalPlayer.Character)

    if not humanoid then
        return false
    end

    local ok, state = pcall(function()
        return humanoid:GetState()
    end)

    if not ok then
        return false
    end

    return state == Enum.HumanoidStateType.Jumping
        or state == Enum.HumanoidStateType.Freefall
end

local AttackRemote
local FallRemote

local function findRemotes()
    local systems = ReplicatedStorage:FindFirstChild("Systems")

    if not systems then
        return
    end

    local actions = systems:FindFirstChild("ActionsSystem")

    if actions then
        local network = actions:FindFirstChild("Network")

        if network then
            AttackRemote = network:FindFirstChild("Attack") or AttackRemote
        end
    end

    local combat = systems:FindFirstChild("CombatSystem")

    if combat then
        local network = combat:FindFirstChild("Network")

        if network then
            FallRemote = network:FindFirstChild("FallDamage") or FallRemote
        end
    end

    if AttackRemote and not AttackRemote.Parent then
        AttackRemote = nil
    end

    if FallRemote and not FallRemote.Parent then
        FallRemote = nil
    end
end

findRemotes()

task.spawn(function()
    for _ = 1, 20 do
        if AttackRemote and FallRemote then
            break
        end

        findRemotes()
        task.wait(0.5)
    end
end)

    if AttackRemote and AttackRemote.Parent then

    end

    return nil
end

local function getESPHealthColor(percent)
    if percent > 0.6 then
        return Color3.fromRGB(60, 255, 110)
    elseif percent > 0.3 then
        return Color3.fromRGB(255, 220, 60)
    end

    return Color3.fromRGB(255, 60, 60)

local partsCache = setmetatable({}, { __mode = "k" })

local function getCharacterParts(character)
    local now = os.clock()
    local cached = partsCache[character]

    if cached and now - cached.Time < 0.5 then
        return cached.Parts
    end

    local parts = {}

    for _, object in ipairs(character:GetDescendants()) do
        if object:IsA("BasePart") and object.Transparency < 1 then
            table.insert(parts, object)
        end
    end

    partsCache[character] = {
        Parts = parts,
        Time = now,
    }

    return parts
end

local BOX_CORNERS = {
    Vector3.new(-1, -1, -1),
    Vector3.new(-1, -1, 1),
    Vector3.new(-1, 1, -1),
    Vector3.new(-1, 1, 1),
    Vector3.new(1, -1, -1),
    Vector3.new(1, -1, 1),
    Vector3.new(1, 1, -1),
    Vector3.new(1, 1, 1),
}

local function getHybridBoundingBox(character)
    if not character or not Camera then
        return nil
    end

    local parts = getCharacterParts(character)

    if #parts == 0 then
        return nil
    end

    local minX = math.huge
    local minY = math.huge
    local maxX = -math.huge
    local maxY = -math.huge
    local visiblePoints = 0

    for _, part in ipairs(parts) do
        if part.Parent then
            local cframe = part.CFrame
            local half = part.Size / 2

            for _, corner in ipairs(BOX_CORNERS) do
                local worldPosition = cframe:PointToWorldSpace(corner * half)
                local screenPosition = Camera:WorldToViewportPoint(worldPosition)

                if screenPosition.Z > 0 then
                    visiblePoints += 1
                    minX = math.min(minX, screenPosition.X)
                    minY = math.min(minY, screenPosition.Y)
                    maxX = math.max(maxX, screenPosition.X)
                    maxY = math.max(maxY, screenPosition.Y)
                end
            end
        end
    end

    if visiblePoints < 2 then
        return nil
    end

    local width = maxX - minX
    local height = maxY - minY

    if width < 2 or height < 2 then
        return nil
    end

    return {
        X = minX,
        Y = minY,
        Width = width,
        Height = height,
        CenterX = (minX + maxX) / 2,
        CenterY = (minY + maxY) / 2,
    }
end

local function hideSkeletonESP(esp)
    if not esp or not esp.Skeleton then
        return
    end

    for _, line in ipairs(esp.Skeleton) do
        line.Visible = false
    end
end

local function hideESP(esp)
    if not esp then
        return
    end

    esp.Box.Visible = false
    esp.NameLabel.Visible = false
    esp.HealthLabel.Visible = false
    esp.DistanceLabel.Visible = false
    esp.Tracer.Visible = false
    esp.HealthBar.Visible = false
    hideSkeletonESP(esp)

    if esp.Chams then
        esp.Chams.Enabled = false
    end

    if esp.ChamsGlow then
        esp.ChamsGlow.Enabled = false
    end

    esp.Gui.Enabled = false
    esp.Mode = "None"
end

local function removeESP(player)
    local esp = ESPObjects[player]

    if not esp then
        return
    end

    safeDestroy(esp.Box)
    safeDestroy(esp.NameLabel)
    safeDestroy(esp.HealthLabel)
    safeDestroy(esp.DistanceLabel)
    safeDestroy(esp.Tracer)
    safeDestroy(esp.HealthBar)
    safeDestroy(esp.Chams)
    safeDestroy(esp.ChamsGlow)

    if esp.Skeleton then
        for _, line in ipairs(esp.Skeleton) do
            safeDestroy(line)
        end
    end

    safeDestroy(esp.Gui)
    safeDestroy(esp.Part)

    ESPObjects[player] = nil
end

local function updateESPTracer(esp, screenX, screenY)
    if not State.ESPTracers or not Camera then
        esp.Tracer.Visible = false
        return
    end

    local viewport = Camera.ViewportSize
    local origin = Vector2.new(viewport.X / 2, viewport.Y)
    local delta = Vector2.new(screenX, screenY) - origin
    local length = delta.Magnitude

    if length < 2 then
        esp.Tracer.Visible = false
        return
    end

    esp.Tracer.Visible = true
    esp.Tracer.BackgroundColor3 = State.ESPColor
    esp.Tracer.Position = UDim2.fromOffset(
        (origin.X + screenX) / 2,
        (origin.Y + screenY) / 2
    )
    esp.Tracer.Size = UDim2.fromOffset(length, 1)
    esp.Tracer.Rotation = math.deg(math.atan2(delta.Y, delta.X))
end

local function updateSkeletonESP(esp, character)
    hideSkeletonESP(esp)

    if not State.ESPSkeleton or not Camera or not character then
        return
    end

    local connections

    if character:FindFirstChild("UpperTorso") then
        connections = R15_SKELETON
    else
        connections = R6_SKELETON
    end

    local lineIndex = 1

    for _, connection in ipairs(connections) do
        local fromPart = character:FindFirstChild(connection[1])
        local toPart = character:FindFirstChild(connection[2])

        if fromPart
            and toPart
            and fromPart:IsA("BasePart")
            and toPart:IsA("BasePart") then

            local fromScreen, fromOnScreen =
                Camera:WorldToViewportPoint(fromPart.Position)
            local toScreen, toOnScreen =
                Camera:WorldToViewportPoint(toPart.Position)

            if fromOnScreen
                and toOnScreen
                and fromScreen.Z > 0
                and toScreen.Z > 0 then

                local deltaX = toScreen.X - fromScreen.X
                local deltaY = toScreen.Y - fromScreen.Y
                local length = math.sqrt(
                    deltaX * deltaX + deltaY * deltaY
                )
                local line = esp.Skeleton[lineIndex]

                if line and length >= 2 then
                    line.Position = UDim2.fromOffset(
                        (fromScreen.X + toScreen.X) / 2,
                        (fromScreen.Y + toScreen.Y) / 2
                    )
                    line.Size = UDim2.fromOffset(length, 2)
                    line.Rotation = math.deg(math.atan2(deltaY, deltaX))
                    line.BackgroundColor3 = State.ESPColor
                    line.Visible = true
                end

                lineIndex += 1

                if lineIndex > #esp.Skeleton then
                    break
                end
            end
        end
    end
end

local function updateScreenESP(player, esp, boundingBox, humanoid, distance)
    esp.Mode = "Screen"
    esp.Gui.Enabled = false

    local character = player.Character
    local chams = esp.Chams
    local chamsGlow = esp.ChamsGlow

    if State.Chams and character then
        if not chams then
            chams = Instance.new("Highlight")
            chams.Name = "AestheticChams"
            chams.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            chams.FillTransparency = 0.2
            chams.OutlineTransparency = 0
            chams.Parent = ESPFolder
            esp.Chams = chams
        end

        if not chamsGlow then
            chamsGlow = Instance.new("Highlight")
            chamsGlow.Name = "AestheticChamsGlow"
            chamsGlow.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            chamsGlow.FillTransparency = 0.82
            chamsGlow.OutlineTransparency = 0.1
            chamsGlow.Parent = ESPFolder
            esp.ChamsGlow = chamsGlow
        end

        chams.Adornee = character
        chams.FillColor = State.ESPColor:Lerp(Color3.new(1, 1, 1), 0.2)
        chams.OutlineColor = State.ESPColor
        chams.Enabled = true

        chamsGlow.Adornee = character
        chamsGlow.FillColor =
            State.ESPColor:Lerp(Color3.new(1, 1, 1), 0.5)
        chamsGlow.OutlineColor = State.ESPColor
        chamsGlow.Enabled = true
    else
        if chams then
            chams.Enabled = false
        end

        if chamsGlow then
            chamsGlow.Enabled = false
        end
    end

    local box = esp.Box
    box.Visible = State.ESPShowBox
    box.Position = UDim2.fromOffset(boundingBox.X, boundingBox.Y)
    box.Size = UDim2.fromOffset(boundingBox.Width, boundingBox.Height)
    box.BackgroundColor3 = State.ESPColor
    box.BackgroundTransparency = 0.9

    esp.BoxStroke.Color = State.ESPColor

    local name = esp.NameLabel
    name.Visible = State.ESPShowName
    name.Text = player.DisplayName
    name.Position = UDim2.fromOffset(boundingBox.CenterX, boundingBox.Y - 3)
    name.Size = UDim2.fromOffset(220, 20)

    local hp = esp.HealthLabel
    hp.Visible = State.ESPShowHealth
    hp.Position = UDim2.fromOffset(
        boundingBox.X + boundingBox.Width + 6,
        boundingBox.CenterY
    )
    hp.Size = UDim2.fromOffset(92, 18)

    local percent = nil

    if humanoid and humanoid.MaxHealth > 0 then
        local health = math.clamp(humanoid.Health, 0, humanoid.MaxHealth)
        local maxHealth = math.max(humanoid.MaxHealth, 1)
        percent = math.clamp(health / maxHealth, 0, 1)

        hp.Text = string.format(
            "%d / %d HP",
            math.floor(health),
            math.floor(maxHealth)
        )
        hp.TextColor3 = getESPHealthColor(percent)
    else
        hp.Text = "? HP"
        hp.TextColor3 = Color3.fromRGB(180, 180, 180)
    end

    local healthBar = esp.HealthBar
    local showHealthBar = State.ESPShowHealth and percent ~= nil

    healthBar.Visible = showHealthBar

    if showHealthBar then
        healthBar.Position = UDim2.fromOffset(boundingBox.X - 8, boundingBox.Y)
        healthBar.Size = UDim2.fromOffset(4, boundingBox.Height)
        esp.HealthBarFill.Size = UDim2.fromScale(1, percent)
        esp.HealthBarFill.BackgroundColor3 = getESPHealthColor(percent)
    end

    local distanceLabel = esp.DistanceLabel
    distanceLabel.Visible = State.ESPShowDistance
    distanceLabel.Position = UDim2.fromOffset(
        boundingBox.CenterX,
        boundingBox.Y + boundingBox.Height + 4
    )
    distanceLabel.Size = UDim2.fromOffset(90, 18)
    distanceLabel.Text = string.format("%d Stud", math.floor(distance))

    updateESPTracer(esp, boundingBox.CenterX, boundingBox.CenterY)
    updateSkeletonESP(esp, character)
end

local function updateFallbackESP(player, esp, distance)
    local position = esp.LastPos

    if not position then
        return false
    end

    esp.Box.Visible = false
    esp.NameLabel.Visible = false
    esp.HealthLabel.Visible = false
    esp.DistanceLabel.Visible = false
    esp.HealthBar.Visible = false
    hideSkeletonESP(esp)

    if esp.Chams then
        esp.Chams.Enabled = false
    end

    if esp.ChamsGlow then
        esp.ChamsGlow.Enabled = false
    end

    if esp.Mode ~= "Fallback" then
        esp.Part.Position = position
        esp.Mode = "Fallback"
    else
        esp.Part.Position = esp.Part.Position:Lerp(position, 0.25)
    end

    esp.Gui.Enabled = true

    esp.FallbackName.Visible = State.ESPShowName
    esp.FallbackName.Text = player.DisplayName

    esp.FallbackDistance.Visible = State.ESPShowDistance
    esp.FallbackDistance.Text = string.format("%d Stud", math.floor(distance))

    local humanoid = getHumanoid(player.Character)
    local showHealth = State.ESPShowHealth
        and humanoid ~= nil
        and humanoid.MaxHealth > 0

    esp.FallbackHealth.Visible = showHealth

    if showHealth then
        local percent = math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)

        esp.FallbackHealthFill.Size = UDim2.fromScale(percent, 1)
        esp.FallbackHealthFill.BackgroundColor3 = getESPHealthColor(percent)
    end

    if Camera then
        local screenPosition, onScreen = Camera:WorldToViewportPoint(position)

        if onScreen and screenPosition.Z > 0 then
            updateESPTracer(esp, screenPosition.X, screenPosition.Y)
        else
            esp.Tracer.Visible = false
        end
    else
        esp.Tracer.Visible = false
    end

    return true
end

track(RunService.RenderStepped:Connect(function()
    if not State.ESPEnabled then
        for _, esp in pairs(ESPObjects) do
            hideESP(esp)
        end

        return
    end

    local localRoot = getRoot(LocalPlayer.Character)

    if not localRoot then
        return
    end

    local now = os.clock()

    for player, esp in pairs(ESPObjects) do
        if player.Parent ~= Players then
            removeESP(player)
            continue
        end

        local shown = false
        local skip = State.TeamCheck and isTeammate(player)
        local character = player.Character
        local humanoid = getHumanoid(character)

        if not skip
            and character
            and humanoid
            and humanoid.Health > 0 then

            local position = getCharacterPosition(character)

            if position then
                local distance = (localRoot.Position - position).Magnitude

                if distance <= State.ESPMaxDistance then
                    esp.LastPos = position
                    esp.LastValidTime = now

                    local boundingBox = getHybridBoundingBox(character)

                    if boundingBox then
                        updateScreenESP(
                            player,
                            esp,
                            boundingBox,
                            humanoid,
                            distance
                        )

                        shown = true
                    end
                end
            end
        end

        if shown then
            continue
        end

        if not skip and esp.LastPos then
            local distance = (localRoot.Position - esp.LastPos).Magnitude

            if distance <= State.ESPMaxDistance then
                if updateFallbackESP(player, esp, distance) then
                    continue
                end
            end
        end

        hideESP(esp)
    end
end))

local function isTargetInsideFOV(character)
    if not State.KillAuraFOV then
        return true
    end

    if not Camera then
        return false
    end

    local aimPart = getAimPart(character)

    if not aimPart then
        return false
    end

    local viewport = Camera.ViewportSize
    local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
    local screenPosition, onScreen =
        Camera:WorldToViewportPoint(aimPart.Position)

    if not onScreen or screenPosition.Z <= 0 then
        return false
    end

    local screenDistance = (
        Vector2.new(screenPosition.X, screenPosition.Y) - center
    ).Magnitude

    return screenDistance <= State.FOVSize
end

local function getNearestTarget()
    local myRoot = getRoot(LocalPlayer.Character)

    if not myRoot then
        return nil
    end

    local nearest = nil
    local nearestDistance = State.KillAuraRange

    for _, player in ipairs(Players:GetPlayers()) do
        if isTargetable(player) then
            local character = player.Character
            local root = getRoot(character)

            if root then
                local distance = (root.Position - myRoot.Position).Magnitude

                if distance <= nearestDistance
                    and isTargetInsideFOV(character) then

                    nearest = character
                    nearestDistance = distance
                end
            end
        end
    end

    return nearest
end

local function getAimbotTarget()
    if not Camera then
        return nil
    end

    local viewport = Camera.ViewportSize
    local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
    local maxDistance = State.AimbotFOVCheck and State.FOVSize or math.huge
    local nearestPart = nil
    local nearestDistance = maxDistance

    for _, player in ipairs(Players:GetPlayers()) do
        if isTargetable(player) then
            local character = player.Character
            local aimPart = getAimPart(character)

            if aimPart then
                local screenPosition, onScreen =
                    Camera:WorldToViewportPoint(aimPart.Position)

                if onScreen and screenPosition.Z > 0 then
                    local screenDistance = (
                        Vector2.new(screenPosition.X, screenPosition.Y)
                        - center
                    ).Magnitude

                    if screenDistance <= nearestDistance then
                        nearestDistance = screenDistance
                        nearestPart = aimPart
                    end
                end
            end
        end
    end

    return nearestPart
end

local function getTargetESPCharacter()
    if not Camera then
        return nil
    end

    local myRoot = getRoot(LocalPlayer.Character)

    if not myRoot then
        return nil
    end

    local viewport = Camera.ViewportSize
    local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
    local maxWorldDistance = math.max(State.ESPMaxDistance, 1)
    local bestCharacter = nil
    local bestScreenDistance = math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        if isTargetable(player) then
            local character = player.Character
            local aimPart = getAimPart(character)

            if aimPart then
                local worldDistance = (
                    myRoot.Position - aimPart.Position
                ).Magnitude

                if worldDistance <= maxWorldDistance then
                    local screenPosition, onScreen =
                        Camera:WorldToViewportPoint(aimPart.Position)

                    if onScreen and screenPosition.Z > 0 then
                        local screenDistance = (
                            Vector2.new(screenPosition.X, screenPosition.Y)
                            - center
                        ).Magnitude

                        local insideFOV = not State.FOVEnabled
                            or screenDistance <= State.FOVSize

                        if insideFOV
                            and screenDistance < bestScreenDistance then

                            bestScreenDistance = screenDistance
                            bestCharacter = character
                        end
                    end
                end
            end
        end
    end

    return bestCharacter
end

        TargetGui.Enabled = false

    local target = getTargetESPCharacter()

        TargetGui.Enabled = false
        return

    local boundingBox = getHybridBoundingBox(target)

    if not boundingBox then
        TargetGui.Enabled = false
        return

    local pad = 4
    local x = boundingBox.X - pad
    local y = boundingBox.Y - pad
    local width = boundingBox.Width + pad * 2
    local height = boundingBox.Height + pad * 2
    local pulse = 0.5 + 0.5 * math.sin(os.clock() * 6)
    local length = math.clamp(math.min(width, height) * 0.18, 8, 18)
    local thickness = 2
    local alpha = 0.1 + pulse * 0.35

    local layouts = {
        { x, y, length, thickness },
        { x, y, thickness, length },
        { x + width - length, y, length, thickness },
        { x + width - thickness, y, thickness, length },
        { x, y + height - thickness, length, thickness },
        { x, y + height - length, thickness, length },
        { x + width - length, y + height - thickness, length, thickness },
        { x + width - thickness, y + height - length, thickness, length },

    for index, corner in ipairs(TargetCorners) do
        local layout = layouts[index]

        corner.Position = UDim2.fromOffset(layout[1], layout[2])
        corner.Size = UDim2.fromOffset(layout[3], layout[4])

        corner.Visible = true

    TargetLabel.Text = player
        and ("TARGET  " .. player.DisplayName)
        or "TARGET"
    TargetLabel.Position = UDim2.fromOffset(
        boundingBox.CenterX,
        y - 20
    )
    TargetLabel.BackgroundTransparency = 0.15 + pulse * 0.2
    TargetLabel.Visible = true

local function invokeAttack(target)
    local remote = getAttackRemote()

    if not remote or not target then
        return false
    end

    local character = LocalPlayer.Character

    if not character then
        return false
    end

    local slot = character:GetAttribute("ReplicatedHotbarSlot")

    if slot == nil then
        return false
    end

    return pcall(function()
        if remote:IsA("RemoteFunction") then
            remote:InvokeServer(target, tostring(slot))
        else
            remote:FireServer(target, tostring(slot))
        end
    end)
end

local function performAttack()
    return invokeAttack(getNearestTarget())
end

local flyRebuildAt = 0

track(RunService.RenderStepped:Connect(function()
    if not State.FlyEnabled or not Camera then
        return
    end

    local character = LocalPlayer.Character
    local root = getRoot(character)

    if not root then
        return
    end

    local bodyVelocity = State.FlyBodyVelocity
    local bodyGyro = State.FlyBodyGyro

    if not bodyVelocity
        or not bodyGyro
        or bodyVelocity.Parent ~= root
        or bodyGyro.Parent ~= root then

        local now = os.clock()

        if now - flyRebuildAt >= 0.5 then
            flyRebuildAt = now
            enableFly()
        end

        return
    end

    if UserInputService:GetFocusedTextBox() then
        bodyVelocity.Velocity = Vector3.zero
        return
    end

    local direction = Vector3.zero

    if UserInputService:IsKeyDown(Enum.KeyCode.W) then
        direction += Camera.CFrame.LookVector
    end

    if UserInputService:IsKeyDown(Enum.KeyCode.S) then
        direction -= Camera.CFrame.LookVector
    end

    if UserInputService:IsKeyDown(Enum.KeyCode.A) then
        direction -= Camera.CFrame.RightVector
    end

    if UserInputService:IsKeyDown(Enum.KeyCode.D) then
        direction += Camera.CFrame.RightVector
    end

    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
        direction += Vector3.new(0, 1, 0)
    end

    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
        direction -= Vector3.new(0, 1, 0)
    end

    if FlyButtons.Forward.Held then
        direction += Camera.CFrame.LookVector
    end

    if FlyButtons.Back.Held then
        direction -= Camera.CFrame.LookVector
    end

    if FlyButtons.Left.Held then
        direction -= Camera.CFrame.RightVector
    end

    if FlyButtons.Right.Held then
        direction += Camera.CFrame.RightVector
    end

    if FlyButtons.Up.Held then
        direction += Vector3.new(0, 1, 0)
    end

    if FlyButtons.Down.Held then
        direction -= Vector3.new(0, 1, 0)
    end

    if direction.Magnitude > 0 then
        bodyVelocity.Velocity = direction.Unit * State.FlySpeed
    else
        bodyVelocity.Velocity = Vector3.zero
    end

    local lookVector = Camera.CFrame.LookVector

    bodyGyro.CFrame = CFrame.lookAt(
        root.Position,
        root.Position + lookVector
    )
end))

                Brightness = Lighting.Brightness,
                ClockTime = Lighting.ClockTime,
                ExposureCompensation = Lighting.ExposureCompensation,

                EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,

        end

        pcall(function()
            Lighting.Ambient = Color3.new(1, 1, 1)
            Lighting.OutdoorAmbient = Color3.new(1, 1, 1)
            Lighting.Brightness = 2
            Lighting.ClockTime = 12
            Lighting.FogEnd = State.Fog and State.FogDistance or 1e9
            Lighting.ExposureCompensation = 0
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0

                Lighting[property] = value
            end)
        end

        fullbrightSaved = nil

        pcall(function()
            Lighting.FogEnd =
                State.Fog and State.FogDistance or defaultFogEnd
        end)
    end
end

end

local ignoredMaterials = State.IgnoreMaterials
local materialList = {}

pcall(function()

    if compiled then
        for _, material in ipairs(compiled:GetChildren()) do
            local name = material.Name

                end

            end
        end
    end

local function getMiningLookPart()
    if not Camera then
        return nil
    end

    local viewport = Camera.ViewportSize
    local ray = Camera:ViewportPointToRay(
        viewport.X / 2,
        viewport.Y / 2
    )

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude

    local character = LocalPlayer.Character

    if character then
        params.FilterDescendantsInstances = { character }
    end

    local result = workspace:Raycast(
        ray.Origin,
        ray.Direction * 20,
        params
    )

    if result then
        return result.Instance
    end

    return nil
end

local function isMaterialIgnored(part)
    if not part or not part:IsA("BasePart") then
        return false
    end

    local variant = part.MaterialVariant

    if not variant or variant == "" then
        return false
    end

    return ignoredMaterials[variant] == true
end

local function getBreakingTimeFunction()
    local systems = ReplicatedStorage:FindFirstChild("Systems")
    local items = systems and systems:FindFirstChild("ItemsSystem")
    local libs = items and items:FindFirstChild("Libs")
    local module = libs and libs:FindFirstChild("GetBreakingTime")

    if not module then
        return nil
    end

    local ok, result = pcall(require, module)

    if ok and type(result) == "function" then
        return result
    end

    return nil
end

local function enableFastBreak()
    if fastBreakHooked then
        return true
    end

    if typeof(hookfunction) ~= "function" then
        return false
    end

    local getBreakingTime = getBreakingTimeFunction()

    if not getBreakingTime then
        return false
    end

    local oldFunction
    local success = pcall(function()
        oldFunction = hookfunction(getBreakingTime, function(...)
            local part = getMiningLookPart()

            if isMaterialIgnored(part) and oldFunction then
                return oldFunction(...)
            end

            return State.BreakSpeed
        end)
    end)

    if not success then
        return false
    end

    fastBreakOldFunction = oldFunction
    fastBreakHooked = true

    return true
end

local function disableFastBreak()
    if not fastBreakHooked then
        return
    end

    if typeof(hookfunction) == "function" and fastBreakOldFunction then
        local getBreakingTime = getBreakingTimeFunction()

        if getBreakingTime then
            pcall(function()
                hookfunction(getBreakingTime, fastBreakOldFunction)
            end)
        end
    end

    fastBreakOldFunction = nil
    fastBreakHooked = false
end

local function hasFS()
    return type(isfile) == "function"
        and type(readfile) == "function"
        and type(writefile) == "function"
        and type(isfolder) == "function"
        and type(makefolder) == "function"
end

local function getImageObj()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")

    if not playerGui then
        return nil
    end

    local cameraGui = playerGui:FindFirstChild("CameraGui")

    if not cameraGui then
        return nil
    end

    local crosshair = cameraGui:FindFirstChild("Crosshair")

    if not crosshair then
        return nil
    end

    if crosshair:IsA("ImageLabel") or crosshair:IsA("ImageButton") then
        return crosshair
    end

    return crosshair:FindFirstChildWhichIsA("ImageLabel", true)
        or crosshair:FindFirstChildWhichIsA("ImageButton", true)
end

local function applyCrosshairImage(id)
    local object = getImageObj()

    if not object then
        return false
    end

    id = tostring(id or ""):match("^%s*(.-)%s*$")

    if id == "" then
        return false
    end

    if not id:match("^rbxassetid://") then
        id = "rbxassetid://" .. id
    end

    object.Image = id

    return true
end

local function applyCrosshairSize(size)
    CrosshairState.Size = math.clamp(tonumber(size) or 32, 8, 200)

    local object = getImageObj()

    if not object then
        return
    end

    local cameraGui = PlayerGui:FindFirstChild("CameraGui")
    local crosshair = cameraGui and cameraGui:FindFirstChild("Crosshair")
    local scale = crosshair and crosshair:FindFirstChildWhichIsA("UIScale", true)

    if scale then
        scale.Scale = CrosshairState.Size / 32
    else
        object.Size = UDim2.fromOffset(
            CrosshairState.Size,
            CrosshairState.Size
        )
    end
end

local function applyCrosshairVisible(value)
    CrosshairState.Enabled = value == true

    if object then
        object.Visible = CrosshairState.Enabled
    end
end

local function getCursorNames()
    local names = {}

    for name, data in pairs(CrosshairState.SavedCursors) do
        if type(data) == "table" and data.id then
            table.insert(names, name)
        end
    end

    table.sort(names)

local function loadCursors()
    CrosshairState.SavedCursors = {}

    if not hasFS() then
        return
    end

    pcall(function()
        if not isfolder(CURSOR_FOLDER) then
            makefolder(CURSOR_FOLDER)
        end

        if not isfile(CURSOR_FILE) then
            writefile(CURSOR_FILE, "{}")
            return
        end

        local raw = readfile(CURSOR_FILE)

        if raw == "" then
            return
        end

        local data = HttpService:JSONDecode(raw)

        if type(data) == "table" then
            CrosshairState.SavedCursors = data
        end
    end)
end

local function saveCursors()
    if not hasFS() then
        return false
    end

    return pcall(function()
        if not isfolder(CURSOR_FOLDER) then
            makefolder(CURSOR_FOLDER)
        end

        writefile(
            CURSOR_FILE,
            HttpService:JSONEncode(CrosshairState.SavedCursors)
        )
    end)
end

loadCursors()

local function getConfigNames()
    local names = {}
    local seen = {}

    if not hasFS() then
        return names
    end

    pcall(function()
        if not isfolder(CONFIG_FOLDER) then
            makefolder(CONFIG_FOLDER)
        end

        local manifestPath = CONFIG_FOLDER .. "/manifest.json"

        if isfile(manifestPath) then
            local raw = readfile(manifestPath)

            if raw ~= "" then
                local data = HttpService:JSONDecode(raw)

                if type(data) == "table" then
                    for _, name in ipairs(data) do
                        if type(name) == "string" and name ~= "" then
                            names[name] = true
                        end
                    end
                end
            end
        end

        if type(listfiles) == "function" then
            local ok, files = pcall(listfiles, CONFIG_FOLDER)

            if ok and type(files) == "table" then
                for _, path in ipairs(files) do
                    local fileName = tostring(path):match("([^/\\]+)%.json$")

                    if fileName
                        and fileName ~= "manifest"
                        and fileName ~= "cursors" then

                        names[fileName] = true
                    end
                end
            end
        end
    end)

    local result = {}

    for name in pairs(names) do
        table.insert(result, name)
    end

    table.sort(result)

    return result
end

            HttpService:JSONEncode(list)
        )
    end)
end

local function readColor(value, fallback)
    if typeof(value) == "Color3" then
        return value
    end

    if type(value) == "table" then
        return Color3.new(
            math.clamp(tonumber(value[1]) or fallback.R, 0, 1),
            math.clamp(tonumber(value[2]) or fallback.G, 0, 1),
            math.clamp(tonumber(value[3]) or fallback.B, 0, 1)
        )
    end

    return fallback
end

local function clampNumber(value, fallback, minimum, maximum)
    local number = tonumber(value)

    if number == nil then
        number = fallback
    end

    return math.clamp(number, minimum, maximum)
end

    State.InfiniteJump = snap.InfiniteJump == true
    State.GodMode = snap.GodMode == true
    State.Fullbright = snap.Fullbright == true
    State.Fog = snap.Fog == true
    State.FogColor = readColor(snap.FogColor, State.FogColor)
    State.FogDistance = clampNumber(snap.FogDistance, 1000, 100, 5000)
    State.ClickTP = snap.ClickTP == true
    State.BunnyHop = snap.BunnyHop == true
    State.Gravity = clampNumber(snap.Gravity, DEFAULT_GRAVITY, 0, 500)

    State.FlyEnabled = snap.FlyEnabled == true
    State.FlySpeed = clampNumber(snap.FlySpeed, 16, 1, 500)
    State.Speed = clampNumber(snap.Speed, 16, 1, 500)
    State.JumpPower = clampNumber(snap.JumpPower, 27, 1, 500)

    State.FastBreak = snap.FastBreak == true
    State.BreakSpeed = clampNumber(snap.BreakSpeed, 0, 0, 3600)

        CrosshairState.Enabled = snap.CrosshairEnabled ~= false
    end

        CrosshairState.Size = clampNumber(snap.CrosshairSize, 32, 8, 200)
    end

                State.IgnoreMaterials[name] = enabled == true
            end
        end

    State.ESPEnabled = false
    State.FOVEnabled = false
    State.KillAura = false
    State.JumpAttack = false
    State.AimbotEnabled = false
    State.FlyEnabled = false

    State.Gravity = DEFAULT_GRAVITY

    pcall(disableFly)
    pcall(disableGodMode)
    pcall(function()
        applyFullbright(false)
    end)
    pcall(function()
        applyFog(false)
    end)
    pcall(applyGravity)
    pcall(function()
        applyNoClip(false)
    end)

    if State.OldNamecall then
        pcall(function()
            hookmetamethod(game, "__namecall", State.OldNamecall)
        end)
    end

    disconnectAll()

    for player in pairs(ESPObjects) do
        removeESP(player)
    end

    safeDestroy(ESPGui)
    safeDestroy(FOVGui)
    safeDestroy(TargetGui)
    safeDestroy(FlyGui)
    safeDestroy(ESPFolder)

    local destroyed = pcall(function()
        Rayfield:Destroy()
    end)

    if not destroyed then
        for _, gui in ipairs(PlayerGui:GetChildren()) do
            if gui:IsA("ScreenGui")
                and tostring(gui.Name):lower():find("rayfield") then

                safeDestroy(gui)
            end
        end
    end

    print("[aesthetic] unloaded")
end

TabConfig:CreateButton({
    Name = "Create Config",
    Callback = function()
        local name = string.match(
            tostring(ConfigNameInput or ""),
            "^%s*(.-)%s*$"
        )

        if name == ""
            or name == "manifest"
            or name == "cursors"
            or name:find("[/\\]") then

            Rayfield:Notify({
                Title = "Config",
                Content = "Enter a valid config name.",
                Duration = 3,
            })

            return
        end

        if not hasFS() then
            Rayfield:Notify({
                Title = "Config",
                Content = "File system unavailable.",
                Duration = 3,
            })

            return
        end

        local path = CONFIG_FOLDER .. "/" .. name .. ".json"
        local success = pcall(function()
            if not isfolder(CONFIG_FOLDER) then
                makefolder(CONFIG_FOLDER)
            end

            writefile(
                path,
                HttpService:JSONEncode(buildStateSnapshot())
            )

            local list = getConfigNames()
            local exists = false

            for _, existing in ipairs(list) do
                if existing == name then
                    exists = true
                    break
                end
            end

            if not exists then
                table.insert(list, name)
            end

            saveManifest(list)
        end)

        if not success then
            Rayfield:Notify({
                Title = "Config",
                Content = "Failed to create config.",
                Duration = 3,
            })

            return
        end

        SelectedConfig = name
        refreshConfigDropdown()

        Rayfield:Notify({
            Title = "Config Created",
            Content = name,
            Duration = 2,
        })
    end,
})

TabConfig:CreateButton({
    Name = "Load Selected Config",
    Callback = function()
        if not SelectedConfig then
            Rayfield:Notify({
                Title = "Config",
                Content = "Select a config first.",
                Duration = 3,
            })

            return
        end

        if not hasFS() then
            return
        end

        local path = CONFIG_FOLDER .. "/" .. SelectedConfig .. ".json"

        if not isfile(path) then
            Rayfield:Notify({
                Title = "Config",
                Content = "File not found.",
                Duration = 3,
            })

            return
        end

        local success, snap = pcall(function()
            return HttpService:JSONDecode(readfile(path))
        end)

        if not success then
            Rayfield:Notify({
                Title = "Config",
                Content = "Invalid JSON.",
                Duration = 3,
            })

            return
        end

        if not applySnapshot(snap) then
            Rayfield:Notify({
                Title = "Config",
                Content = "Failed to apply config.",
                Duration = 3,
            })

            return
        end

        Rayfield:Notify({
            Title = "Config Loaded",
            Content = SelectedConfig,
            Duration = 2,
        })
    end,
})
